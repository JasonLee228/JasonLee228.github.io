```java
package com.bookstudy.springboot.domain.team;

import lombok.Builder;
import lombok.Getter;
import lombok.NoArgsConstructor;

import javax.persistence.*;

@Getter
@NoArgsConstructor
@Entity
public class Team {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(length = 500, nullable = false)
    private String teamname;

    private String leader;

    @Column(columnDefinition = "TEXT", nullable = true)
    private String introdution;

    @Builder
    public Team(String teamname, String leader, String introdution)
    {
        this.teamname = teamname;
        this.leader = leader;
        this.introdution = introdution;
    }

    public void update(String teamname, String leader, String introdution) {
        this.teamname = teamname;
        this.leader = leader;
        this.introdution = introdution;
    }
}
```

```java
package com.bookstudy.springboot.service.team;

import com.bookstudy.springboot.domain.team.Team;
import com.bookstudy.springboot.domain.team.TeamRepository;
import com.bookstudy.springboot.web.dto.*;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;
import java.util.stream.Collectors;

@RequiredArgsConstructor
@Service
public class teamService {

    private final TeamRepository teamRepository;

    @Transactional
    public Long save(TeamSaveRequestDto requestDto) {
        return teamRepository.save(requestDto.toEntity()).getId();
    }

    public TeamResponseDto findById(Long id) {
        Team entity = teamRepository.findById(id).orElseThrow(()
                -> new IllegalArgumentException("해당 게시글이 없습니다. id : " + id));

        return new TeamResponseDto(entity);
    }

    @Transactional
    public void delete(Long id) {
        Team team = teamRepository.findById(id).orElseThrow(() -> new IllegalArgumentException("해당 게시글이 없습니다. id : " + id));
        teamRepository.delete(team);
    }

    @Transactional(readOnly = true)
    public List<TeamResponseDto> findAllUser() {
        return teamRepository.findAllDesc().stream()
                .map(TeamResponseDto::new)
                .collect(Collectors.toList());
    }
}
```

#### #TeamApiController

{% raw %}
```java
package com.bookstudy.springboot.web;


import com.bookstudy.springboot.service.posts.PostsService;
import com.bookstudy.springboot.service.team.teamService;
import com.bookstudy.springboot.web.dto.*;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.*;

@RequiredArgsConstructor
@RestController
public class TeamApiController {

    private final PostsService postsService;
    private final teamService teamService;

    @PostMapping("/api/v1/team")
    public Long save(@RequestBody TeamSaveRequestDto requestDto) {
        return teamService.save(requestDto);
    }

    @PutMapping("/api/v1/team/{id}")
    //@PostMapping("/api/v1/posts/")
    public Long update(@PathVariable Long id, @RequestBody PostsUpdateRequestDto requestDto) {
        return postsService.update(id, requestDto);
    }

    @GetMapping("/api/v1/team/{id}")
    public TeamResponseDto findId(@PathVariable Long id) {
        return teamService.findById(id);
    }

    @DeleteMapping("/api/v1/team/{id}")
    public Long delete(@PathVariable Long id) {
        teamService.delete(id);
        return id;
    }

    
}
```

#### #team.js 추가(index.js에서 컬럼명만 수정)

{% raw %}
```text
var main = {
    init : function () {
        var _this = this;
        $('#btn-save').on('click', function () {
            _this.save();
        });
         $('#btn-update').on('click', function () {
            _this.update();
        });
        $('#btn-delete').on('click', function () {
            _this.delete();
        });
    },
    save : function () {
        var data = {
            teamname: $('#teamname').val(),
            leader: $('#leader').val(),
            introdution: $('#introdution').val()
        };

        $.ajax({
            type: 'POST',
            url: '/api/v1/team',
            dataType: 'json',
            contentType:'application/json; charset=utf-8',
            data: JSON.stringify(data)
        }).done(function() {
            alert('팀이 등록되었습니다.');
            window.location.href = '/';
        }).fail(function (error) {
            alert(JSON.stringify(error));
        });
    },
     update : function () {
            var data = {
                title: $('#title').val(),
                content: $('#content').val()
            };

            var id = $('#id').val();

            $.ajax({
                type: 'PUT',
                url: '/api/v1/team/'+id,
                dataType: 'json',
                contentType:'application/json; charset=utf-8',
                data: JSON.stringify(data)
            }).done(function() {
                alert('글이 수정되었습니다.');
                window.location.href = '/';
            }).fail(function (error) {
                alert(JSON.stringify(error));
            });
        },
     delete : function() {
        var id = $('#id').val();

        $.ajax({
            type: 'DELETE',
            url: '/api/v1/team/'+id,
            dataType: 'json',
            contentType:'application/json; charset=utf-8'
        }).done(function() {
            alert('글이 삭제되었습니다.');
            window.location.href = '/';
        }).fail(function (error) {
            alert(JSON.stringify(error));
        });
     }
};

main.init();
```

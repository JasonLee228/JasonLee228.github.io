문제 

> 💡 명령을 실행하는 소프트웨어를 작성하라. 명령은 다음과 같이 mov,add, sub, jn0, prt로 구성되며 mov sum 0은 sum변수에 0을 삽입하고, add sum i는 sum+i 이고, sub n 1은 n값을 1 감소시킨다 그리고 jn0 n 3은 n이 0이 아니면 3번째 줄로 돌아가라는 의미이다. 
> prt sum 0은 sum의 내용을 출력하고 종료한다.
> go를 입력하면 지금까지 입력한 명령어를 실행한다.
> 추가적으로, stop을 입력하게 되면 프로그램을 종료하도록 했다.

```java
package chapter7;
import java.util.*;
class Instruction {
	
	private String line;//명령어 한줄 담을 line
	private String opcode;// ADD와 같은 명령 담을 변수
	private String operand [] = new String [2];//명령 안의 실행요소들 담을
	public Instruction(String line) {
		this.line = line;
		line = line.toUpperCase(); // 대문자로 만들기
		StringTokenizer st = new StringTokenizer(line);
		// 	line이 "ADD S J"
		opcode = st.nextToken(); // 첫 토큰, 명령, "ADD"
		operand[0] = st.nextToken(); // 변수 담길 0번째
		operand[1] = st.nextToken(); // 사용값, 계산될 다른 변수나 수가 담길 1번째
	}
	public String getOpcode() {
		return opcode;//명령어 꺼내올 메소드
	}
	public String getOperand(int index) {
		if(index < 0 || index > 2)
			return null;
		return operand[index];
	}
	public String toString() {
		return "[" + line + "] ";
	}
}

class machine
{//해쉬맵은 사용자가 정의한 변수들, mov명령이 실행될 때 변수를 담는 용도로 사용, 벡터는 명령어 한줄 전체를 담는 용도.
	private HashMap<String, Integer> memory = new HashMap<String, Integer>();
	private Vector<Instruction> program = new Vector<Instruction>();
	private Scanner scanner = new Scanner(System.in);

	public machine() {
		System.out.println("수퍼컴이 작동합니다. go를 입력하면 실행합니다.");
	}
	public void startCom() {
		while(true)
		{
			if(!run())return;
			instruction_value();
			clear();
		}
	}
	public boolean run() {//boolean형으로 만들어서 사용자가 사용종료를 입력하면 false를 리턴, 종료하도록 하였다.
		String line;
		while(true)
		{
			System.out.print((program.size()) + ">> ");
			line = scanner.nextLine();
			if(line.equals("go"))
			{
				break;
			}
			if(line.equals("stop"))
			{
				System.out.println("System off!");
				return false;
			}
			program.add(new Instruction(line));
		}
		return true;
		
	}
	
	public void instruction_value() {
		int pc = 0;//다음 명령어 저장할 변수 pc
		while(pc<program.size())
		{
			Instruction inst = program.get(pc);//첫 번째 명령부터 받아와서 실행
			pc++;//미리 다음 명령 준비
			switch (inst.getOpcode()) {//opcode에는 명령라인의 첫 번째 토큰인 명령이 담겨있다.
			case "MOV":
				System.out.println("mov명령 실행됨");
				mov(inst);//각각 메소드 실행할 때마다 현재 실행되고 있는 명령의 한줄을 넣어준다.
				break;
			case "ADD":
				System.out.println("add명령 실행됨");
				add(inst);
				break;
			case "SUB":
				System.out.println("sub명령 실행됨");
				sub(inst);
				break;
			case "JN0":
				System.out.println("jn0명령 실행됨");
				int newpc = jn0(inst);
				if(newpc == -1)//그대로 진행
					break;
				else {
					pc = newpc;//되돌아가기
					break;
				}
			case "PRT":
				try {

					System.out.println("prt명령 실행됨");
					prt(inst);
					break;
				} catch (NullPointerException e) {//명령어를 입력하지 않은 채로 없는 명령어의 출력을 하려 할 때
					System.out.println("정의되어있는 명령어가 없습니다!");
				}
			default:
				error(pc-1);//잘못된 명령어 입력 시 에러난 부분을 표시해준다.
				break;
			}
		}
		clear();
		
	}
	private void error(int errPc) {
		System.out.println(errPc + "줄에서 오류 발생!");
	}
	public void clear() {
		program.removeAllElements(); // 벡터의 모든 요소 삭제
		memory.clear();

	}
	public int getvalue(String op) {//변수인 경우와 실수인 경우
		int n;
		try {
			n = Integer.parseInt(op);
		} catch (NumberFormatException e) {
			n = memory.get(op);//간접주소라고 보면 될 것 같은데, 명령어의 세 번째 토큰인 계산할 수 지정 부분에
		}//숫자가 아닌 다른 변수가 들어있을 경우 사용되는 부분임. 해쉬맵에서 그 변수의 값을 가져와 넘긴다.
		
		return n;
		
	}
	private void mov(Instruction instruction) {
	
		String key = instruction.getOperand(0);
		String value = instruction.getOperand(1);
		int n = getvalue(value);
		memory.put(key, n);
	}
	
	public void add(Instruction instruction) {
		String key = instruction.getOperand(0);
		String value = instruction.getOperand(1);
		int n = getvalue(value);
		int m = memory.get(key);
		memory.put(key, m+n);
	}
	
	public void sub(Instruction instruction) {
		String key = instruction.getOperand(0);
		String value = instruction.getOperand(1);
		int n = getvalue(value);
		int m = memory.get(key);
		memory.put(key, m-n);
	}
	
	public int jn0(Instruction instruction) {
		String op = instruction.getOperand(0);//getoperand의 0번째에는 jn0명령어로 검사할 변수가 담겨있다
		int ad = Integer.parseInt(instruction.getOperand(1));
		int n = memory.get(op);//1번째에는 돌아갈 주소(줄)이 담겨있으니 그 부분으로 돌아가도록 한다.
		if(n!=0)
			return ad;
		else {//0일 경우에는 돌아갈 부분이 없으니 -1 리턴
			return -1;
		}
		
	}
	
	public void prt(Instruction instruction) {//현재까지 저장된 변수들의 출력 이후 종료되는 메소드
		instruction.toString();//prt명령의 한줄 출력을 위해 실행된 메소드
		
		Set<String> keys = memory.keySet();
		Iterator<String> it = keys.iterator();
		//위 두줄은 해쉬맵의 출력을 위해서
		while(it.hasNext())
		{
			String oprand = it.next();
			int value = memory.get(oprand);
			System.out.print(oprand + " : " + value + "   ");
		}
		System.out.println();
		int ad = memory.get(instruction.getOperand(0));
		System.out.println("출력할 결과는 " + ad + ". 프로그램 실행 끝");
		
	}
	
}


public class Ex13 {

	public static void main(String[] args) {
		// TODO Auto-generated method stub
		machine mycomputer = new machine();
		mycomputer.startCom();
	}

}
```

            try
            {        
                if(note_write.charAt(0)=='0')
                {
                    //br.close();  //이거때문에 안되네 왜지 음 음음

                else{
                    
                    notepad.add( (notepad.size()+1) + ". "+ nickname.get(index)+ " : " + note_write);
                    //viewNote();//한줄한줄 쓸 때마다 보여주는 것보다 다 쓰고 보여주는게 낫다고 생각해서 바꿈

                        System.out.println("한글자 이상 적으셔야 합니다. 다시 적어주세요.");
                       
                    }
        }
        
    }

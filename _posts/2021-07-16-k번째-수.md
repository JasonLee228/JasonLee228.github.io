```java
import java.util.ArrayList;
import java.util.*;

class Solution {
    public int[] solution(int[] array, int[][] commands) {
        int row = commands.length;//1.
        int[] answer = new int[row];//2.
        int[] temp;
     
         int index=0,tempindex=0;
        for(int i=0;i<row;i++)//3.
        {
            temp = new int[commands[i][1]-commands[i][0]+1];//4.
            tempindex=0;
            for(int k=commands[i][0]-1;k<commands[i][1];k++)//5.
            {
                temp[tempindex]=array[k];//6.
                tempindex++;//7.
            }

            Arrays.sort(temp);//8.
            answer[index] = temp[commands[i][2]-1];//9.
            index++;//10.
        }
        return answer;
    }
}
```
